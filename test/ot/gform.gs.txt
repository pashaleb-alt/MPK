function createQuiz() {
  var form = FormApp.create("Тест: " + DATA.title);
  form.setIsQuiz(true)
      .setDescription(DATA.source + "\n\nВыберите один ответ в каждом вопросе. После отправки покажется результат и пояснения.")
      .setCollectEmail(false).setLimitOneResponsePerUser(false).setShowLinkToRespondAgain(true)
      .setProgressBar(true).setShuffleQuestions(false).setConfirmationMessage("Ответы приняты. Нажмите «Посмотреть результат».");
  form.addTextItem().setTitle("Фамилия, имя, отчество").setRequired(true);
  form.addTextItem().setTitle("Должность, подразделение").setRequired(true);
  form.addDateItem().setTitle("Дата прохождения теста").setRequired(true);
  form.addPageBreakItem().setTitle("Вопросы");
  DATA.items.forEach(function (it, i) {
    var item = form.addMultipleChoiceItem().setTitle((i + 1) + ". " + it.q).setRequired(true).setPoints(1);
    item.setChoices(it.opts.map(function (o, k) { return item.createChoice(o, k === it.right); }));
    item.setFeedbackForCorrect(FormApp.createFeedback().setText(it.exp).build());
    item.setFeedbackForIncorrect(FormApp.createFeedback().setText("Верно: " + it.opts[it.right] + ". " + it.exp).build());
  });
  var ss = SpreadsheetApp.create("Результаты — " + DATA.title);
  form.setDestination(FormApp.DestinationType.SPREADSHEET, ss.getId());
  addResultsSheet(ss, DATA.items.length);
  Logger.log("Форма для прохождения: " + form.getPublishedUrl());
  Logger.log("Редактор формы:        " + form.getEditUrl());
  Logger.log("Таблица результатов:   " + ss.getUrl());
}

// Лист «Результаты»: ФИО, должность, дата, баллы, %, оценка — формулами поверх листа ответов
function addResultsSheet(ss, n) {
  SpreadsheetApp.flush();
  var sh = ss.insertSheet("Результаты", 0);
  var scale = ss.insertSheet("Шкала");
  scale.getRange("A1:C5").setValues([["Порог, % (от)", "Оценка", "Пояснение"],
    [0, "Не сдан", "менее 60 %"], [60, "Удовлетворительно", "60–74 %"], [75, "Хорошо", "75–89 %"], [90, "Отлично", "90 % и выше"]]);
  scale.getRange("A1:C1").setFontWeight("bold");
  var src = "'Ответы на форму (1)'";
  sh.getRange("A1:G1").setValues([["Отправлено", "ФИО", "Должность", "Дата теста", "Баллы", "%", "Оценка"]]).setFontWeight("bold");
  // Лист ответов: A=время, B=баллы («x / n»), C=ФИО, D=должность, E=дата, далее вопросы
  sh.getRange("A2").setFormula('=IFERROR(QUERY(' + src + '!A2:E, "select A, C, D, E where A is not null", 0), "")');
  sh.getRange("E2").setFormula('=ARRAYFORMULA(IF(A2:A="", "", VALUE(REGEXEXTRACT(TO_TEXT(' + src + '!B2:B), "^\\d+"))))');
  sh.getRange("F2").setFormula('=ARRAYFORMULA(IF(A2:A="", "", ROUND(E2:E/' + n + '*100)))');
  sh.getRange("G2").setFormula('=ARRAYFORMULA(IF(A2:A="", "", VLOOKUP(F2:F, Шкала!A2:B5, 2, TRUE)))');
  sh.setColumnWidths(1, 1, 150); sh.setColumnWidths(2, 2, 220); sh.setFrozenRows(1);
  sh.getRange("A2:A").setNumberFormat("dd.mm.yyyy hh:mm"); sh.getRange("D2:D").setNumberFormat("dd.mm.yyyy");
  var rules = [
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo("Отлично").setBackground("#c6efce").setRanges([sh.getRange("G2:G")]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo("Хорошо").setBackground("#e2f0d9").setRanges([sh.getRange("G2:G")]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo("Удовлетворительно").setBackground("#ffeb9c").setRanges([sh.getRange("G2:G")]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo("Не сдан").setBackground("#ffc7ce").setRanges([sh.getRange("G2:G")]).build()];
  sh.setConditionalFormatRules(rules);
}
